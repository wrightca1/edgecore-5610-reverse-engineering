# CPLD Register Dump (eLBC CS1 @ 0xEA000000)
# Captured from Cumulus 2.5.1, March 27, 2026
# 32-byte repeating pattern (CPLD has 32 unique registers)
#
# Offset : Value : Description
0x00 = 0x10  # CPLD version (1.0)
0x01 = 0x00  # (reserved)
0x02 = 0x02  # Board revision
0x03 = 0x10  # Board model ID
0x04 = 0x00  # (reserved)
0x05 = 0x03  # PSU status (both PSUs present)
0x06 = 0xEA  # (read-back of base address?)
0x07 = 0xEA  # (read-back)
0x08 = 0x0F  # Fan tray present (all 4 fans)
0x09 = 0x07  # Fan tray status (3 OK?)
0x0A = 0xAA  # LED control register
0x0B = 0xFA  # LED status
0x0C = 0xFA  # LED status 2
0x0D = 0x08  # Fan PWM speed (8 = ~25%, range 0-31)
0x0E = 0x00  # (reserved/interrupt)
0x0F = 0xFA  # (status)
0x10 = 0xFF  # SFP TX disable bitmap (all disabled = 0xFF?)
0x11 = 0xFF  # SFP TX disable bitmap 2
0x12 = 0x00  # SFP present bitmap
0x13 = 0x52  # Board model string? ('R' = 0x52)
0x14 = 0xFF  # SFP RX LOS bitmap
0x15 = 0x01  # QSFP control
0x16 = 0xEA  # (read-back)
0x17 = 0xEA  # (read-back)
0x18 = 0xEA  # (read-back)
0x19 = 0xEA  # (read-back)
0x1A = 0xEA  # (read-back)
0x1B = 0xEA  # (read-back)
0x1C = 0x08  # (mirrors 0x0D?)
0x1D = 0xEA  # (read-back)
0x1E = 0xEA  # (read-back)
0x1F = 0xEA  # (read-back)
